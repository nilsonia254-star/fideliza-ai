document.addEventListener("DOMContentLoaded", carregarClientes); async function carregarClientes() { try { let resposta = await fetch('https://n8n.nodestack.shop/webhook-test/fidelidade-mary'); let clientes = await resposta.json(); let select = document.getElementById('cliente-selecao'); select.innerHTML = ''; clientes.forEach(c => { let opt = document.createElement('option'); opt.value = c.telefone; opt.textContent = c.nome; select.appendChild(opt); }); } catch (erro) { console.log("Erro ao carregar clientes:", erro); document.getElementById('cliente-selecao').innerHTML = ''; } } function mudarClienteAdmin() { let select = document.getElementById('cliente-selecao'); let nomeAdm = document.getElementById('nome-adm'); let telAdm = document.getElementById('tel-adm'); if (select.value) { nomeAdm.textContent = select.options[select.selectedIndex].text; telAdm.textContent = select.value; } else { nomeAdm.textContent = "-"; telAdm.textContent = "-"; } }
